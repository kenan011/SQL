--SUAL1

SELECT FIRST_NAME || ' ' || SUBSTR(LAST_NAME,1,1) || '.' AS QISA_AD,
        LOWER(EMAIL) || '@company.com'  AS EMAIL,
        LENGTH(FIRST_NAME) AS AD_UZUNLUQU
FROM HR.EMPLOYEES 
ORDER BY AD_UZUNLUQU DESC,QISA_AD ;

--SUAL2

SELECT FIRST_NAME,SALARY,
        SALARY * 12 * (1 + NVL(COMMISSION_PCT,0)) AS ILLIK_GELIR,
        CASE WHEN SALARY * 12 * (1 + NVL(COMMISSION_PCT,0)) >= 150000 THEN 'Yuksek'
             WHEN SALARY * 12 * (1 + NVL(COMMISSION_PCT,0)) >= 100000 THEN 'Orta'
             ELSE 'Asagi'
        END AS SEVIYYE
FROM HR.EMPLOYEES
ORDER BY ILLIK_GELIR DESC;

--SUAL3

SELECT JOB_ID,
        COUNT(*) AS ISCI_SAYI,
        MIN(SALARY) AS EN_AZ,
        MAX(SALARY) AS EN_COX,
        MAX(SALARY) - MIN(SALARY) AS FERQ
FROM HR.EMPLOYEES
GROUP BY JOB_ID
HAVING COUNT(*) >= 5
ORDER BY ISCI_SAYI DESC;

--SUAL 4

SELECT M.FIRST_NAME || ' ' || M.LAST_NAME AS MENECER,
        COUNT(*) AS TABE_SAYI,
        ROUND(AVG(E.SALARY)) AS TABE_ORTASI
FROM HR.EMPLOYEES E
JOIN HR.EMPLOYEES M
ON E.MANAGER_ID = M.EMPLOYEE_ID
GROUP BY M.FIRST_NAME || ' ' || M.LAST_NAME 
HAVING ROUND(AVG(E.SALARY)) >  8500
ORDER BY TABE_ORTASI DESC;

SELECT FIRST_NAME || ' ' || LAST_NAME AS ISCI,
        TO_CHAR(HIRE_DATE,'DD.MM.YYYY') AS ISE_GIRIS,
        TO_CHAR(HIRE_DATE,'Day') AS GUN,
        TRUNC(MONTHS_BETWEEN(SYSDATE,HIRE_DATE)/ 12) STAJ_IL,
        TO_CHAR(ADD_MONTHS(HIRE_DATE,6),'DD.MM.YYYY') AS SINAQ_BITIR
FROM HR.EMPLOYEES
WHERE HIRE_DATE >= TO_DATE('01.01.2007','DD.MM.YYYY')
ORDER BY HIRE_DATE;


--SUAL7

SELECT FIRST_NAME,DEPARTMENT_ID,SALARY,SOBE_ORTASI,FERQ,YER
FROM (SELECT FIRST_NAME,
             DEPARTMENT_ID,
             SALARY,
             COUNT(*) OVER(PARTITION BY DEPARTMENT_ID) AS ISCI_SAYI
             ROUND(AVG(SALARY) OVER(PARTITION BY DEPARTMENT_ID)) AS SOBE_ORTASI,
             (SALARY - ROUND(AVG(SALARY))OVER(PARTITION BY DEPARTMENT_ID)) AS FERQ,
             DENSE_RANK() OVER(PARTITION BY DEPARTMENT_ID ORDER BY SALARY ASC) AS YER
        FROM HR.EMPLOYEES)
WHERE ISCI_SAYI >= 3
ORDER BY DEPARTMENT_ID,YER;
