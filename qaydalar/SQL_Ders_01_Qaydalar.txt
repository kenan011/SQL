-- ============================================================
-- SQL DƏRS 1: QAYDALAR VƏ NÜMUNƏLƏR
-- Bütün nümunələr Oracle-da HR sxemi üzərində yazılıb.
-- İki defislə (--) başlayan sətirlər şərhdir, işlədilmir.
-- ============================================================


-- ============================================================
-- 1. SELECT, *, DISTINCT, ALIAS
-- ============================================================

-- * bütün sütunları gətirir
SELECT * FROM HR.EMPLOYEES;

-- Yalnız lazım olan sütunları yazmaq daha düzgündür
SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES;

-- SQL əmrləri və sütun adları böyük-kiçik hərfə həssas deyil.
-- Yəni depARTment_id və DEPARTMENT_ID eynidir.
-- DISTINCT təkrar sətirləri silir.
-- İki sütun yazılanda həmin iki sütunun BİRLİKDƏ təkrarlanan birləşməsini silir.
SELECT DISTINCT FIRST_NAME, depARTment_id FROM HR.EMPLOYEES;

-- Bir sütun üzrə unikal dəyərlər
SELECT DISTINCT DEPARTMENT_ID FROM HR.EMPLOYEES;

-- ALIAS: sütuna müvəqqəti ad vermək (AS ilə).
-- Hesablanmış sütunlara ad vermək üçün çox rahatdır.
SELECT FIRST_NAME, SALARY * 12 AS ILLIK_MAAS FROM HR.EMPLOYEES;


-- ============================================================
-- 2. WHERE: sətirləri süzmək
-- Operatorlar: =, >, <, >=, <=, <>, !=, AND, OR, BETWEEN, IN, LIKE
-- ============================================================

-- = bərabərdir
SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE DEPARTMENT_ID = 50;

-- <> və != eyni mənanı verir: bərabər deyil
SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE DEPARTMENT_ID <> 50;

SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE DEPARTMENT_ID != 50;

-- AND: hər iki şərt doğru olmalıdır
SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE DEPARTMENT_ID >= 50 AND DEPARTMENT_ID <= 80;

-- BETWEEN: aralıq, hər iki sərhəd daxildir.
-- Bu, DEPARTMENT_ID >= 50 AND DEPARTMENT_ID <= 100 ilə eynidir.
SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE DEPARTMENT_ID BETWEEN 50 AND 100;

-- OR: şərtlərdən biri doğru olsa kifayətdir
SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE DEPARTMENT_ID = 50 OR DEPARTMENT_ID = 100;

-- IN: siyahıdakı dəyərlərdən biri. Yuxarıdakı OR ilə eynidir, amma daha qısadır.
-- Diqqət: BETWEEN 50 AND 100 aralığı gətirir, IN (50, 100) isə yalnız 50 və 100-ü.
SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE DEPARTMENT_ID IN (50, 100);

-- AND əvvəl hesablanır, ona görə OR ilə birlikdə mötərizə yazın
SELECT FIRST_NAME, SALARY, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE SALARY > 10000 AND (DEPARTMENT_ID = 80 OR DEPARTMENT_ID = 90);

-- Mətn TƏK dırnaqla yazılır və böyük-kiçik hərfə HƏSSASDIR.
-- 'Steven' nəticə verir, 'steven' vermir.
SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE FIRST_NAME = 'Steven';


-- ============================================================
-- 3. LIKE: mətn üzrə axtarış
-- %  istənilən sayda simvol (sıfır da ola bilər)
-- _  tam olaraq bir simvol
--
-- 'S%'    S ilə başlayan
-- '%a'    a ilə bitən
-- '%an%'  içində an olan
-- 'S_'    S ilə başlayan, 2 simvollu
-- '__'    tam 2 simvollu
-- 'A___'  A ilə başlayan, 4 simvollu
-- ============================================================

SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE FIRST_NAME LIKE 'S%';

-- Kiçik a ilə bitən adlar. '%A' yazsaq, böyük A axtarılar və nəticə çıxmaz.
SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE FIRST_NAME LIKE '%a';

SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE FIRST_NAME LIKE '__';

SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE FIRST_NAME LIKE 'A___';

-- SƏHV: = ilə % işləmir. Bu sorğu xəta vermir, amma adı məhz "S%" olan işçini axtarır
-- və heç nə qaytarmır. % və _ yalnız LIKE ilə işləyir.
SELECT FIRST_NAME, DEPARTMENT_ID FROM HR.EMPLOYEES
WHERE FIRST_NAME = 'S%';


-- ============================================================
-- 4. NULL: dəyərin olmaması
-- NULL sıfır deyil, boş dəyərdir. = ilə yoxlanılmır.
-- ============================================================

SELECT FIRST_NAME, COMMISSION_PCT FROM HR.EMPLOYEES
WHERE COMMISSION_PCT IS NULL;

SELECT FIRST_NAME, COMMISSION_PCT FROM HR.EMPLOYEES
WHERE COMMISSION_PCT IS NOT NULL;

-- SƏHV: = NULL xəta vermir, amma heç vaxt sətir qaytarmır
SELECT FIRST_NAME, COMMISSION_PCT FROM HR.EMPLOYEES
WHERE COMMISSION_PCT = NULL;


-- ============================================================
-- 5. ORDER BY: sıralamaq
-- ASC  artan sıra (standart, yazılmaya da bilər)
-- DESC azalan sıra
-- ============================================================

SELECT FIRST_NAME, SALARY FROM HR.EMPLOYEES
ORDER BY SALARY;

SELECT FIRST_NAME, SALARY FROM HR.EMPLOYEES
ORDER BY SALARY DESC;

-- Bir neçə sütun: əvvəl şöbəyə görə, eyni şöbədə isə maaşa görə
SELECT FIRST_NAME, DEPARTMENT_ID, SALARY FROM HR.EMPLOYEES
ORDER BY DEPARTMENT_ID ASC, SALARY DESC;

-- Alias ORDER BY-da işləyir
SELECT FIRST_NAME, SALARY * 12 AS ILLIK_MAAS FROM HR.EMPLOYEES
ORDER BY ILLIK_MAAS DESC;

-- DESC sıralamada NULL dəyərlər yuxarı çıxır. NULLS LAST onları sona salır.
SELECT FIRST_NAME, COMMISSION_PCT FROM HR.EMPLOYEES
ORDER BY COMMISSION_PCT DESC NULLS LAST;


-- ============================================================
-- 6. FETCH FIRST: ilk N sətri götürmək
-- Yazılış: FETCH FIRST 1 ROW ONLY və ya FETCH FIRST 5 ROWS ONLY
-- ORDER BY-dan sonra yazılır. ORDER BY olmadan "ilk 5" təsadüfi ola bilər.
-- ============================================================

-- Ən yüksək maaşlı 5 işçi
SELECT FIRST_NAME, SALARY FROM HR.EMPLOYEES
ORDER BY SALARY DESC
FETCH FIRST 5 ROWS ONLY;

-- Ən aşağı maaşlı işçi
SELECT FIRST_NAME, SALARY FROM HR.EMPLOYEES
ORDER BY SALARY ASC
FETCH FIRST 1 ROW ONLY;


-- ============================================================
-- 7. AGGREGATE FUNKSİYALAR: çox sətirdən bir nəticə
-- COUNT say, SUM cəm, AVG orta, MIN ən kiçik, MAX ən böyük
-- NULL dəyərləri nəzərə almırlar. Yalnız COUNT(*) bütün sətirləri sayır.
-- ============================================================

SELECT COUNT(SALARY),
       SUM(SALARY),
       AVG(SALARY),
       MIN(SALARY),
       MAX(SALARY)
FROM HR.EMPLOYEES;

-- COUNT(*) bütün sətirlər, COUNT(sütun) NULL olmayanlar,
-- COUNT(DISTINCT sütun) unikal dəyərlər
SELECT COUNT(*) AS BUTUN_SETIRLER,
       COUNT(COMMISSION_PCT) AS KOMISSIYASI_OLANLAR,
       COUNT(DISTINCT DEPARTMENT_ID) AS SOBE_SAYI
FROM HR.EMPLOYEES;

-- WHERE ilə: yalnız 50-ci şöbə üçün
SELECT COUNT(*) AS SAY, AVG(SALARY) AS ORTA FROM HR.EMPLOYEES
WHERE DEPARTMENT_ID = 50;

-- SƏHV: adi sütunu aggregate funksiya ilə yanaşı yazmaq olmaz.
-- ORA-00937 xətası verir. Bunun həlli növbəti dərsdə keçiləcək GROUP BY-dır.
-- SELECT DISTINCT DEPARTMENT_ID, AVG(SALARY) FROM HR.EMPLOYEES;


-- ============================================================
-- 8. ROUND VƏ FLOOR: yuvarlaqlaşdırma
-- ROUND(x, 2)   vergüldən sonra 2 rəqəm
-- ROUND(x)      tam ədədə qədər
-- ROUND(x, -3)  minliklərə qədər (məsələn, 6461 -> 6000)
-- FLOOR(x)      həmişə aşağıya, tam ədədə qədər
-- ============================================================

SELECT AVG(SALARY),
       ROUND(AVG(SALARY), 2),
       ROUND(AVG(SALARY)),
       ROUND(AVG(SALARY), -3),
       FLOOR(AVG(SALARY))
FROM HR.EMPLOYEES;


-- ============================================================
-- 9. YADDA SAXLAYIN: sorğunun yazılış və icra sırası
--
-- Yazılış: SELECT -> FROM -> WHERE -> ORDER BY -> FETCH FIRST
-- İcra:    FROM -> WHERE -> SELECT -> ORDER BY -> FETCH FIRST
--
-- Buna görə SELECT-dəki alias WHERE-də işləmir, ORDER BY-da işləyir.
-- SƏHV: WHERE ILLIK_MAAS > 60000
-- DÜZGÜN: WHERE SALARY * 12 > 60000
-- ============================================================

SELECT FIRST_NAME, SALARY * 12 AS ILLIK_MAAS FROM HR.EMPLOYEES
WHERE SALARY * 12 > 60000
ORDER BY ILLIK_MAAS DESC;
