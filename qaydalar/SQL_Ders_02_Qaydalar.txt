-- ============================================================
-- SQL DƏRS 2: QAYDALAR VƏ NÜMUNƏLƏR
-- Mövzular: GROUP BY, HAVING, JOIN (INNER, LEFT, RIGHT, FULL)
-- Bütün nümunələr Oracle-da (FreeSQL) yazılıb.
--
-- HR cədvəllərinin adını həmişə HR. ilə yazın: HR.EMPLOYEES.
-- HR. yazmasanız, Oracle cədvəli öz sxeminizdə axtarır.
-- ============================================================


-- ============================================================
-- 1. GROUP BY: sətirləri qruplara bölmək
--
-- Eyni dəyəri olan sətirlər bir qrupa yığılır.
-- Aggregate funksiya (COUNT, SUM, AVG, MIN, MAX) HƏR QRUP ÜÇÜN ayrıca hesablanır.
-- Nəticədə hər qrup bir sətir olur.
-- ============================================================

-- Dərs 1-də bu sorğu ORA-00937 xətası verirdi:
-- SELECT DEPARTMENT_ID, AVG(SALARY) FROM HR.EMPLOYEES;
-- GROUP BY ilə düzgün işləyir: hər şöbə üçün bir sətir.
SELECT DEPARTMENT_ID, COUNT(*) AS ISCI_SAYI, ROUND(AVG(SALARY), 2) AS ORTA_MAAS
FROM HR.EMPLOYEES
GROUP BY DEPARTMENT_ID
ORDER BY DEPARTMENT_ID;

-- Nəticədə DEPARTMENT_ID-si boş olan bir sətir də görəcəksiniz.
-- Şöbəsi olmayan işçilər ayrıca NULL qrupuna düşür.

-- ƏSAS QAYDA: SELECT-də aggregate funksiyanın içində OLMAYAN hər sütun
-- GROUP BY-da da yazılmalıdır.

-- SƏHV: FIRST_NAME GROUP BY-da yoxdur. ORA-00979 xətası verir.
-- Səbəb: bir şöbədə çoxlu işçi var. Şöbə bir sətir olanda Oracle
-- həmin sətirdə hansı işçinin adını göstərəcəyini bilmir.
-- SELECT FIRST_NAME, DEPARTMENT_ID, AVG(SALARY)
-- FROM HR.EMPLOYEES
-- GROUP BY DEPARTMENT_ID;


-- ------------------------------------------------------------
-- 1.1 İki sütunla GROUP BY
-- Qrup iki sütunun dəyəri BİRLİKDƏ eyni olan sətirlərdən yaranır.
-- ------------------------------------------------------------

SELECT DEPARTMENT_ID, JOB_ID, COUNT(*) AS ISCI_SAYI
FROM HR.EMPLOYEES
GROUP BY DEPARTMENT_ID, JOB_ID
ORDER BY DEPARTMENT_ID;

-- 50-ci şöbəyə baxın: o, 3 sətrə bölünür, çünki orada 3 fərqli vəzifə var:
-- 50  ST_MAN    5
-- 50  ST_CLERK  20
-- 50  SH_CLERK  20
-- Yalnız GROUP BY DEPARTMENT_ID yazsaydıq, 50-ci şöbə bir sətir olardı (45).


-- ============================================================
-- 2. HAVING: qrupları süzmək
--
-- WHERE   SƏTİRLƏRİ süzür, qruplaşdırmadan ƏVVƏL işləyir.
-- HAVING  QRUPLARI süzür, qruplaşdırmadan SONRA işləyir.
--
-- COUNT, SUM, AVG, MIN, MAX ilə yazılan şərt yalnız HAVING-də olur.
-- HAVING həmişə GROUP BY-dan sonra yazılır.
-- ============================================================

-- Orta maaşı 5000-dən çox olan şöbələr
SELECT DEPARTMENT_ID, AVG(SALARY)
FROM HR.EMPLOYEES
GROUP BY DEPARTMENT_ID
HAVING AVG(SALARY) > 5000;

-- 5-dən çox işçisi olan şöbələr
SELECT DEPARTMENT_ID, COUNT(*)
FROM HR.EMPLOYEES
GROUP BY DEPARTMENT_ID
HAVING COUNT(*) > 5;

-- SƏHV: aggregate funksiya WHERE-də işləmir. ORA-00934 xətası verir.
-- Səbəb: WHERE işləyəndə hələ qruplar yoxdur, AVG hesablanmayıb.
-- SELECT DEPARTMENT_ID, AVG(SALARY) FROM HR.EMPLOYEES
-- WHERE AVG(SALARY) > 5000
-- GROUP BY DEPARTMENT_ID;


-- ------------------------------------------------------------
-- 2.1 WHERE ilə HAVING-in fərqi: eyni rəqəm, fərqli nəticə
-- ------------------------------------------------------------

-- A) WHERE: maaşı 5000-dən çox olan hər İŞÇİ saxlanılır, qalanları atılır.
--    Orta maaş yalnız qalan işçilərdən hesablanır.
SELECT DEPARTMENT_ID, COUNT(*), AVG(SALARY)
FROM HR.EMPLOYEES
WHERE SALARY > 5000
GROUP BY DEPARTMENT_ID;

-- B) HAVING: əvvəl hər şöbənin bütün işçilərindən orta maaş hesablanır.
--    Sonra ortası 5000-dən çox olan ŞÖBƏ saxlanılır.
SELECT DEPARTMENT_ID, COUNT(*), AVG(SALARY)
FROM HR.EMPLOYEES
GROUP BY DEPARTMENT_ID
HAVING AVG(SALARY) > 5000;

-- 30-cu şöbəyə baxın. Orada 6 işçi var: biri 11000, qalan beşi 3100 və daha az alır.
-- A sorğusunda 30-cu şöbə görünür: 1 işçi, orta maaş 11000.
-- B sorğusunda görünmür: 6 işçinin orta maaşı 4150-dir.


-- ------------------------------------------------------------
-- 2.2 WHERE və HAVING birlikdə
-- 1) WHERE maaşı 3000 və daha az olan işçiləri atır
-- 2) qalan işçilər şöbələrə bölünür
-- 3) HAVING orta maaşı 5000 və daha az olan şöbələri atır
-- ------------------------------------------------------------

SELECT DEPARTMENT_ID, AVG(SALARY)
FROM HR.EMPLOYEES
WHERE SALARY > 3000
GROUP BY DEPARTMENT_ID
HAVING AVG(SALARY) > 5000;


-- ============================================================
-- 3. SORĞUNUN YAZILIŞ VƏ İCRA SIRASI
--
-- Yazılış: SELECT -> FROM -> JOIN -> WHERE -> GROUP BY -> HAVING -> ORDER BY -> FETCH FIRST
-- İcra:    FROM -> JOIN -> WHERE -> GROUP BY -> HAVING -> SELECT -> ORDER BY -> FETCH FIRST
--
-- Buna görə:
-- - WHERE aggregate funksiyanı görmür, çünki qruplar hələ yaranmayıb.
-- - SELECT-dəki alias WHERE-də işləmir, ORDER BY-da işləyir.
-- - GROUP BY və HAVING-də alias yox, tam ifadəni yazın: HAVING AVG(SALARY) > 5000
-- ============================================================

SELECT DEPARTMENT_ID, COUNT(*) AS ISCI_SAYI
FROM HR.EMPLOYEES
WHERE SALARY > 3000
GROUP BY DEPARTMENT_ID
HAVING COUNT(*) > 5
ORDER BY ISCI_SAYI DESC;


-- ============================================================
-- 4. JOIN: iki cədvəli birləşdirmək
--
-- JOIN iki cədvəlin sətirlərini ON şərtinə görə yan-yana qoyur.
-- ON-da adətən hər iki cədvəldə olan ortaq sütun yazılır, məsələn DEPARTMENT_ID.
--
-- INNER JOIN  yalnız hər iki cədvəldə uyğunu olan sətirlər
-- LEFT JOIN   soldakı cədvəlin BÜTÜN sətirləri + sağdan uyğun olanlar
-- RIGHT JOIN  sağdakı cədvəlin BÜTÜN sətirləri + soldan uyğun olanlar
-- FULL JOIN   hər iki cədvəlin BÜTÜN sətirləri
--
-- Sol cədvəl FROM-dan sonra yazılan, sağ cədvəl JOIN-dən sonra yazılan cədvəldir.
-- Uyğunu tapılmayan tərəfin sütunları NULL (boş) olur.
-- ============================================================

-- Dərsdə istifadə etdiyimiz iki kiçik cədvəl.
-- Əgər hələ yaratmamısınızsa, bu hissəni bütövlükdə işlədin.
-- Hər əmr nöqtəli vergüllə (;) bitməlidir, sonda COMMIT yazılır.

CREATE TABLE demo_departments (
    department_id   NUMBER PRIMARY KEY,
    department_name VARCHAR2(30)
);

CREATE TABLE demo_employees (
    first_name    VARCHAR2(30),
    department_id NUMBER
);

-- Xəta çıxsa:
-- ORA-00955  cədvəl artıq var. Yenidən CREATE etməyə ehtiyac yoxdur.
-- ORA-00001  bu DEPARTMENT_ID ilə sətir artıq var, yəni INSERT ikinci dəfə işlədilib.

SELECT * FROM DEMO_DEPARTMENTS;
-- DEPARTMENT_ID  DEPARTMENT_NAME
-- 10             IT
-- 20             HR
-- 30             Finance          <- bu şöbədə işçi yoxdur

SELECT * FROM DEMO_EMPLOYEES;
-- FIRST_NAME  DEPARTMENT_ID
-- Aysel       10
-- Murad       20
-- Nigar       20
-- Leyla       (null)              <- bu işçinin şöbəsi yoxdur

-- Aşağıdakı sorğularda hər iki DEPARTMENT_ID göstərilir ki, hansı tərəfin boş qaldığı görünsün.
-- Nəticədə birinci DEPARTMENT_ID işçidən (E), ikincisi şöbədən (D) gəlir.


-- ------------------------------------------------------------
-- 4.1 INNER JOIN: yalnız uyğunu olanlar. 3 sətir.
-- JOIN yazmaq INNER JOIN yazmaqla eynidir, INNER sözü yazılmaya bilər.
-- ------------------------------------------------------------

SELECT E.FIRST_NAME, E.DEPARTMENT_ID, D.DEPARTMENT_ID, D.DEPARTMENT_NAME
FROM DEMO_EMPLOYEES E
JOIN DEMO_DEPARTMENTS D
ON E.DEPARTMENT_ID = D.DEPARTMENT_ID;

-- FIRST_NAME  DEPARTMENT_ID  DEPARTMENT_ID  DEPARTMENT_NAME
-- Aysel       10             10             IT
-- Murad       20             20             HR
-- Nigar       20             20             HR
--
-- Leyla düşdü: onun DEPARTMENT_ID-si boşdur. NULL heç bir dəyərə bərabər deyil.
-- Finance düşdü: onun işçisi yoxdur.

-- INNER JOIN-da cədvəllərin sırası nəticəni dəyişmir.
-- Bu sorğu da eyni 3 sətri verir:
SELECT E.FIRST_NAME, E.DEPARTMENT_ID, D.DEPARTMENT_ID, D.DEPARTMENT_NAME
FROM DEMO_DEPARTMENTS D
JOIN DEMO_EMPLOYEES E
ON E.DEPARTMENT_ID = D.DEPARTMENT_ID;


-- ------------------------------------------------------------
-- 4.2 LEFT JOIN: soldakı cədvəlin (DEMO_EMPLOYEES) bütün sətirləri. 4 sətir.
-- LEFT JOIN və LEFT OUTER JOIN eynidir, OUTER sözü yazılmaya bilər.
-- ------------------------------------------------------------

SELECT E.FIRST_NAME, E.DEPARTMENT_ID, D.DEPARTMENT_ID, D.DEPARTMENT_NAME
FROM DEMO_EMPLOYEES E
LEFT JOIN DEMO_DEPARTMENTS D
ON E.DEPARTMENT_ID = D.DEPARTMENT_ID;

-- FIRST_NAME  DEPARTMENT_ID  DEPARTMENT_ID  DEPARTMENT_NAME
-- Aysel       10             10             IT
-- Murad       20             20             HR
-- Nigar       20             20             HR
-- Leyla       (null)         (null)         (null)     <- şöbəsi yoxdur, amma soldadır, ona görə qalır


-- ------------------------------------------------------------
-- 4.3 RIGHT JOIN: sağdakı cədvəlin (DEMO_DEPARTMENTS) bütün sətirləri. 4 sətir.
-- ------------------------------------------------------------

SELECT E.FIRST_NAME, E.DEPARTMENT_ID, D.DEPARTMENT_ID, D.DEPARTMENT_NAME
FROM DEMO_EMPLOYEES E
RIGHT JOIN DEMO_DEPARTMENTS D
ON E.DEPARTMENT_ID = D.DEPARTMENT_ID;

-- FIRST_NAME  DEPARTMENT_ID  DEPARTMENT_ID  DEPARTMENT_NAME
-- Aysel       10             10             IT
-- Murad       20             20             HR
-- Nigar       20             20             HR
-- (null)      (null)         30             Finance    <- işçisi yoxdur, amma sağdadır, ona görə qalır

-- RIGHT JOIN, cədvəllərin yeri dəyişdirilmiş LEFT JOIN-dir.
-- Bu sorğu yuxarıdakı ilə eyni nəticəni verir:
SELECT E.FIRST_NAME, E.DEPARTMENT_ID, D.DEPARTMENT_ID, D.DEPARTMENT_NAME
FROM DEMO_DEPARTMENTS D
LEFT JOIN DEMO_EMPLOYEES E
ON E.DEPARTMENT_ID = D.DEPARTMENT_ID;

-- LEFT və RIGHT JOIN-da sıra vacibdir: cədvəllərin yerini dəyişsəniz, nəticə dəyişir.


-- ------------------------------------------------------------
-- 4.4 FULL JOIN: hər iki cədvəlin bütün sətirləri. 5 sətir.
-- ------------------------------------------------------------

SELECT E.FIRST_NAME, E.DEPARTMENT_ID, D.DEPARTMENT_ID, D.DEPARTMENT_NAME
FROM DEMO_EMPLOYEES E
FULL JOIN DEMO_DEPARTMENTS D
ON E.DEPARTMENT_ID = D.DEPARTMENT_ID;

-- FIRST_NAME  DEPARTMENT_ID  DEPARTMENT_ID  DEPARTMENT_NAME
-- Aysel       10             10             IT
-- Murad       20             20             HR
-- Nigar       20             20             HR
-- Leyla       (null)         (null)         (null)     <- soldan gəlib, sağda uyğunu yoxdur
-- (null)      (null)         30             Finance    <- sağdan gəlib, solda uyğunu yoxdur


-- ============================================================
-- 5. JOIN-də ALIAS VƏ ORA-00918
--
-- Cədvələ qısa ad (alias) verilir: FROM DEMO_EMPLOYEES E
-- Oracle-da cədvəl alias-ı AS olmadan yazılır.
-- Sonra sütunun qarşısına alias yazılır: E.FIRST_NAME, D.DEPARTMENT_NAME
-- ============================================================

-- SƏHV: DEPARTMENT_ID hər iki cədvəldə var. Qarşısında alias olmasa,
-- Oracle hansı cədvəlin sütununu götürəcəyini bilmir. ORA-00918 xətası verir.
-- SELECT FIRST_NAME, DEPARTMENT_ID, DEPARTMENT_NAME
-- FROM DEMO_EMPLOYEES E
-- JOIN DEMO_DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID;

-- DÜZGÜN:
SELECT E.FIRST_NAME, E.DEPARTMENT_ID, D.DEPARTMENT_NAME
FROM DEMO_EMPLOYEES E
JOIN DEMO_DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID;

-- Yalnız bir cədvəldə olan sütunlarda (FIRST_NAME) alias məcburi deyil,
-- amma oxumaq rahat olsun deyə hər sütunun qarşısına alias yazmaq yaxşı vərdişdir.


-- ============================================================
-- 6. LEFT JOIN + IS NULL: uyğunu olmayanları tapmaq
--
-- LEFT JOIN-dən sonra sağ tərəfi boş qalan sətirlər "uyğunu olmayanlar"dır.
-- Onları WHERE ... IS NULL ilə seçirik.
-- ============================================================

-- İşçisi olmayan şöbələr
SELECT D.DEPARTMENT_NAME
FROM DEMO_DEPARTMENTS D
LEFT JOIN DEMO_EMPLOYEES E ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
WHERE E.FIRST_NAME IS NULL;
-- Nəticə: Finance

-- Şöbəsi olmayan işçilər
SELECT E.FIRST_NAME
FROM DEMO_EMPLOYEES E
LEFT JOIN DEMO_DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
WHERE D.DEPARTMENT_ID IS NULL;
-- Nəticə: Leyla


-- ============================================================
-- 7. JOIN + GROUP BY
-- Əvvəl cədvəllər birləşir, sonra birləşmiş nəticə qruplanır.
-- ============================================================

-- Hər şöbənin adı və işçi sayı
SELECT D.DEPARTMENT_NAME, COUNT(*) AS ISCI_SAYI
FROM DEMO_EMPLOYEES E
JOIN DEMO_DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
GROUP BY D.DEPARTMENT_NAME;
-- IT 1, HR 2. Finance yoxdur, çünki INNER JOIN onu atır.

-- DİQQƏT: LEFT JOIN ilə COUNT(*) və COUNT(sütun) fərqli nəticə verir.
-- COUNT(*)            bütün sətirləri sayır, boş sətri də
-- COUNT(E.FIRST_NAME) yalnız işçi adı boş olmayan sətirləri sayır
SELECT D.DEPARTMENT_NAME, COUNT(*) AS SAY_1, COUNT(E.FIRST_NAME) AS SAY_2
FROM DEMO_DEPARTMENTS D
LEFT JOIN DEMO_EMPLOYEES E ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
GROUP BY D.DEPARTMENT_NAME;

-- DEPARTMENT_NAME  SAY_1  SAY_2
-- IT               1      1
-- HR               2      2
-- Finance          1      0      <- Finance-də işçi yoxdur, düzgün cavab 0-dır
--
-- LEFT JOIN-dən sonra Finance üçün bir sətir qalır, amma işçi tərəfi boşdur.
-- COUNT(*) bu boş sətri də sayır. Ona görə sağ cədvəldəki sütunu sayın.


-- ============================================================
-- 8. ÜÇ CƏDVƏL
-- Hər yeni cədvəl üçün ayrıca JOIN ... ON yazılır.
--
-- HR cədvəlləri bu sütunlarla birləşir:
-- HR.EMPLOYEES.DEPARTMENT_ID  = HR.DEPARTMENTS.DEPARTMENT_ID
-- HR.EMPLOYEES.JOB_ID         = HR.JOBS.JOB_ID
-- HR.DEPARTMENTS.LOCATION_ID  = HR.LOCATIONS.LOCATION_ID
-- ============================================================

-- Hər işçinin adı, şöbəsi və şöbənin yerləşdiyi şəhər
SELECT E.FIRST_NAME, E.LAST_NAME, D.DEPARTMENT_NAME, L.CITY
FROM HR.EMPLOYEES E
JOIN HR.DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
JOIN HR.LOCATIONS L ON D.LOCATION_ID = L.LOCATION_ID;

-- HR.EMPLOYEES-də 107 işçi var. Onlardan biri (Kimberely Grant) heç bir şöbəyə bağlı deyil.
-- Ona görə EMPLOYEES ilə DEPARTMENTS arasında INNER JOIN 106, LEFT JOIN 107 sətir qaytarır.

-- 9. YADDA SAXLAYIN
--
-- - SELECT-də aggregate olmayan hər sütun GROUP BY-da olmalıdır.
-- - Sətir şərti WHERE-də, qrup şərti (COUNT, AVG...) HAVING-də yazılır.
-- - JOIN = INNER JOIN. LEFT JOIN = LEFT OUTER JOIN.
-- - LEFT və RIGHT JOIN-da cədvəllərin sırası vacibdir, INNER JOIN-da yox.
-- - İki cədvəldə eyni adlı sütun varsa, qarşısına alias yazın: E.DEPARTMENT_ID
-- - LEFT JOIN-dən sonra saymaq üçün COUNT(*) yox, sağ cədvəlin sütununu sayın.
-- - HR cədvəllərinin qarşısına HR. yazın.
