-- ============================================================
-- SQL DƏRS 3: QAYDALAR VƏ NÜMUNƏLƏR
-- Mövzular: CASE WHEN, NVL, NVL2
-- Bütün nümunələr Oracle-da (FreeSQL) yazılıb.
-- HR cədvəllərinin adını həmişə HR. ilə yazın: HR.EMPLOYEES.
-- ============================================================


-- ============================================================
-- 1. CASE WHEN nədir
--
-- CASE WHEN SQL-in "əgər ... onda ..." (if-else) yazılışıdır.
-- Hər sətir üçün şərtləri yoxlayır və bir dəyər qaytarır.
-- Sətirləri yığmır: 107 sətir girir, 107 sətir çıxır.
--
-- Yazılışı:
-- CASE
--     WHEN şərt_1 THEN dəyər_1
--     WHEN şərt_2 THEN dəyər_2
--     ELSE dəyər_3
-- END AS sutun_adi
--
-- CASE ilə başlayır, END ilə bitir. END-i unutmayın.
-- ============================================================

SELECT FIRST_NAME, SALARY,
       CASE
           WHEN SALARY < 5000  THEN 'Asagi'
           WHEN SALARY < 10000 THEN 'Orta'
           ELSE 'Yuksek'
       END AS MAAS_SEVIYYESI
FROM HR.EMPLOYEES;

-- Necə işləyir (məsələn, maaşı 7000 olan işçi üçün):
-- 7000 < 5000?   Xeyr -> növbəti WHEN
-- 7000 < 10000?  Bəli -> 'Orta' qaytarılır və CASE dayanır
-- İkinci WHEN-də "SALARY >= 5000 AND SALARY < 10000" yazmağa ehtiyac yoxdur,
-- çünki 5000-dən az alanlar artıq birinci WHEN-də tutulub.


-- ============================================================
-- 2. WHEN-LƏRİN SIRASI VACİBDİR
-- Birinci doğru WHEN qalib gəlir, qalanlarına baxılmır.
-- ============================================================

-- SƏHV: birinci şərt çox genişdir, 'Yuksek' heç vaxt çıxmır.
-- 24000 alan Steven King də 'Orta' olur, çünki 24000 > 5000 artıq doğrudur.
SELECT FIRST_NAME, SALARY,
       CASE
           WHEN SALARY > 5000  THEN 'Orta'
           WHEN SALARY > 10000 THEN 'Yuksek'
           ELSE 'Asagi'
       END AS SEVIYYE
FROM HR.EMPLOYEES;

-- DÜZGÜN: ən dar şərt yuxarıda yazılır
SELECT FIRST_NAME, SALARY,
       CASE
           WHEN SALARY > 10000 THEN 'Yuksek'
           WHEN SALARY > 5000  THEN 'Orta'
           ELSE 'Asagi'
       END AS SEVIYYE
FROM HR.EMPLOYEES;

-- Bu səhv xəta vermir, sorğu işləyir. Ona görə nəticəyə baxıb yoxlamaq lazımdır.


-- ============================================================
-- 3. ELSE YAZILMASA
-- Heç bir WHEN tutmayan sətirlərdə NULL (boş) çıxır.
-- ============================================================

SELECT FIRST_NAME, SALARY,
       CASE WHEN SALARY > 10000 THEN 'Yuksek' END AS SEVIYYE
FROM HR.EMPLOYEES;
-- Maaşı 10000 və daha az olanlarda SEVIYYE boşdur.


-- ============================================================
-- 4. WHEN-DƏ İSTƏNİLƏN ŞƏRT YAZILA BİLƏR
-- WHERE-də işlədilən hər şey burada da işləyir:
-- =, <>, >, <, AND, OR, IN, BETWEEN, LIKE, IS NULL
-- ============================================================

SELECT FIRST_NAME, JOB_ID, SALARY, COMMISSION_PCT,
       CASE
           WHEN JOB_ID IN ('AD_PRES', 'AD_VP')           THEN 'Rehberlik'
           WHEN JOB_ID LIKE '%CLERK'                     THEN 'Kargozar'
           WHEN COMMISSION_PCT IS NOT NULL AND SALARY > 10000 THEN 'Guclu satici'
           WHEN COMMISSION_PCT IS NOT NULL               THEN 'Satici'
           WHEN SALARY BETWEEN 8000 AND 12000            THEN 'Mutexessis'
           ELSE 'Diger'
       END AS QRUP
FROM HR.EMPLOYEES;

-- Burada da sıra vacibdir: 'Guclu satici' şərti 'Satici'-dən yuxarıda olmalıdır,
-- yoxsa bütün komissiyalı işçilər 'Satici' olar.

-- NULL yoxlanılarkən həmişə IS NULL / IS NOT NULL yazılır.
-- SƏHV: WHEN COMMISSION_PCT = NULL THEN ... heç vaxt doğru olmur.


-- ============================================================
-- 5. SIMPLE CASE: bir sütunu dəyərlərlə müqayisə etmək
-- Sütunun adı CASE-dən sonra BİR DƏFƏ yazılır.
-- Yalnız bərabərliyi (=) yoxlayır.
-- ============================================================

SELECT FIRST_NAME, DEPARTMENT_ID,
       CASE DEPARTMENT_ID
           WHEN 10 THEN 'Administration'
           WHEN 50 THEN 'Shipping'
           WHEN 80 THEN 'Sales'
           ELSE 'Diger'
       END AS SOBE
FROM HR.EMPLOYEES;

-- Bu, aşağıdakı ilə eynidir, sadəcə qısa yazılışdır:
-- CASE WHEN DEPARTMENT_ID = 10 THEN 'Administration'
--      WHEN DEPARTMENT_ID = 50 THEN 'Shipping' ...

-- >, <, LIKE, IN, IS NULL lazımdırsa, simple CASE işə yaramır, 1-ci bölmədəki yazılış lazımdır.

-- TƏLƏ: simple CASE NULL-u tuta bilmir.
-- CASE COMMISSION_PCT WHEN NULL THEN 'yoxdur' ... heç vaxt 'yoxdur' qaytarmır,
-- çünki NULL = NULL doğru deyil. Bunun yerinə:
SELECT FIRST_NAME,
       CASE WHEN COMMISSION_PCT IS NULL THEN 'yoxdur' ELSE 'var' END AS KOMISSIYA
FROM HR.EMPLOYEES;


-- ============================================================
-- 6. BÜTÜN NƏTİCƏLƏR EYNİ TİPDƏ OLMALIDIR
-- THEN və ELSE-dəki dəyərlərin hamısı ya mətn, ya da rəqəm olmalıdır.
-- ============================================================

-- SƏHV: THEN mətndir ('Yuksek'), ELSE rəqəmdir (0). ORA-00932 xətası verir.
-- SELECT CASE WHEN SALARY > 10000 THEN 'Yuksek' ELSE 0 END FROM HR.EMPLOYEES;

-- DÜZGÜN: ikisi də mətn
SELECT CASE WHEN SALARY > 10000 THEN 'Yuksek' ELSE 'Diger' END AS SEVIYYE
FROM HR.EMPLOYEES;


-- ============================================================
-- 7. ALIAS: TƏK DIRNAQ YOX
--
-- Tək dırnaq '...'  mətn üçündür: 'Yuksek'
-- Cüt dırnaq "..."  ad (sütun adı, alias) üçündür
-- ============================================================

-- SƏHV: alias tək dırnaqda. ORA-00923 xətası verir.
-- SELECT CASE WHEN SALARY > 10000 THEN 'Yuksek' ELSE 'Diger' END AS 'maas seviyyesi'
-- FROM HR.EMPLOYEES;

-- DÜZGÜN 1 (tövsiyə olunur): alt xətt ilə, dırnaqsız
SELECT CASE WHEN SALARY > 10000 THEN 'Yuksek' ELSE 'Diger' END AS MAAS_SEVIYYESI
FROM HR.EMPLOYEES;

-- DÜZGÜN 2: boşluq mütləq lazımdırsa, cüt dırnaqla
SELECT CASE WHEN SALARY > 10000 THEN 'Yuksek' ELSE 'Diger' END AS "maas seviyyesi"
FROM HR.EMPLOYEES;
-- Cüt dırnaqlı alias-a sonra müraciət edəndə də eyni dırnaq və eyni hərflər lazımdır:
-- ORDER BY "maas seviyyesi"


-- ============================================================
-- 8. CASE İLƏ HESABLAMA
-- THEN-dən sonra təkcə mətn yox, hesablama da yazıla bilər.
-- ============================================================

-- Maaş artımı: 5000-dən az alanlara 20%, 5000 ilə 10000 arası alanlara 10%,
-- qalanlarına artım yoxdur.
SELECT FIRST_NAME, SALARY,
       CASE
           WHEN SALARY < 5000   THEN SALARY * 1.20
           WHEN SALARY <= 10000 THEN SALARY * 1.10
           ELSE SALARY
       END AS YENI_MAAS
FROM HR.EMPLOYEES;
-- Bütün nəticələr rəqəmdir, ona görə 6-cı bölmədəki qayda pozulmur.


-- ============================================================
-- 9. CASE VƏ JOIN BİRLİKDƏ
-- CASE JOIN-dən sonra gələn sütunlarla da işləyir.
-- ============================================================

-- Şöbəsi olmayan işçi üçün boş yer əvəzinə 'Sobesiz' yazmaq
SELECT E.FIRST_NAME,
       CASE
           WHEN D.DEPARTMENT_NAME IS NULL THEN 'Sobesiz'
           ELSE D.DEPARTMENT_NAME
       END AS SOBE
FROM DEMO_EMPLOYEES E
LEFT JOIN DEMO_DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID;

-- FIRST_NAME  SOBE
-- Aysel       IT
-- Murad       HR
-- Nigar       HR
-- Leyla       Sobesiz


-- ============================================================
-- 10. CASE ALIAS-I HARADA İŞLƏYİR
--
-- İcra sırası: FROM -> WHERE -> SELECT -> ORDER BY
-- CASE SELECT-də hesablanır. Ona görə:
-- - ORDER BY-da alias işləyir (SELECT-dən sonra gəlir)
-- - WHERE-də alias işləmir (SELECT-dən əvvəl gəlir)
-- ============================================================

-- ORDER BY-da alias işləyir
SELECT FIRST_NAME, SALARY,
       CASE
           WHEN SALARY > 10000 THEN 'Yuksek'
           WHEN SALARY > 5000  THEN 'Orta'
           ELSE 'Asagi'
       END AS SEVIYYE
FROM HR.EMPLOYEES
ORDER BY SEVIYYE, SALARY DESC;
-- Mətn olduğu üçün əlifba sırası ilə düzülür: Asagi, Orta, Yuksek

-- SƏHV: WHERE-də alias işləmir. ORA-00904 xətası verir.
-- SELECT FIRST_NAME, CASE WHEN SALARY > 10000 THEN 'Yuksek' ELSE 'Diger' END AS SEVIYYE
-- FROM HR.EMPLOYEES
-- WHERE SEVIYYE = 'Yuksek';

-- DÜZGÜN: WHERE-də şərtin özünü yazın
SELECT FIRST_NAME, CASE WHEN SALARY > 10000 THEN 'Yuksek' ELSE 'Diger' END AS SEVIYYE
FROM HR.EMPLOYEES
WHERE SALARY > 10000;


-- ============================================================
-- 11. NVL VƏ NVL2: NULL üçün qısa yazılış
--
-- NVL(x, y)       x boşdursa y-ni, boş deyilsə x-in özünü qaytarır
-- NVL2(x, a, b)   x boş DEYİLSƏ a-nı, boşdursa b-ni qaytarır
--
-- İkisi də CASE-in qısa formasıdır:
-- NVL(x, y)      =  CASE WHEN x IS NULL     THEN y ELSE x END
-- NVL2(x, a, b)  =  CASE WHEN x IS NOT NULL THEN a ELSE b END
-- ============================================================

-- ------------------------------------------------------------
-- 11.1 Problem: NULL ilə istənilən hesab NULL verir
-- ------------------------------------------------------------

SELECT FIRST_NAME, SALARY, COMMISSION_PCT,
       SALARY + SALARY * COMMISSION_PCT AS AYLIQ_GELIR
FROM HR.EMPLOYEES;
-- Komissiyası olmayanlarda AYLIQ_GELIR boş çıxır, halbuki maaşları var.
-- Səbəb: SALARY * NULL = NULL, SALARY + NULL = NULL.


-- ------------------------------------------------------------
-- 11.2 NVL: boş yerə dəyər qoymaq
-- ------------------------------------------------------------

SELECT FIRST_NAME, SALARY,
       NVL(COMMISSION_PCT, 0) AS KOMISSIYA,
       SALARY + SALARY * NVL(COMMISSION_PCT, 0) AS AYLIQ_GELIR
FROM HR.EMPLOYEES;
-- İndi komissiyası olmayanlarda komissiya 0 sayılır və AYLIQ_GELIR = SALARY olur.

-- Eyni şey CASE ilə (daha uzun):
-- CASE WHEN COMMISSION_PCT IS NULL THEN 0 ELSE COMMISSION_PCT END

-- NVL mətn sütunu ilə də işləyir. 9-cu bölmədəki CASE-in qısa yazılışı:
SELECT E.FIRST_NAME,
       NVL(D.DEPARTMENT_NAME, 'Sobesiz') AS SOBE
FROM DEMO_EMPLOYEES E
LEFT JOIN DEMO_DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID;
-- Aysel IT, Murad HR, Nigar HR, Leyla Sobesiz


-- ------------------------------------------------------------
-- 11.3 NVL-də iki dəyər eyni tipdə olmalıdır
-- ------------------------------------------------------------

-- SƏHV: COMMISSION_PCT rəqəmdir, 'yoxdur' mətndir. ORA-01722 xətası verir.
-- SELECT FIRST_NAME, NVL(COMMISSION_PCT, 'yoxdur') FROM HR.EMPLOYEES;
--
-- Səbəb: NVL ya COMMISSION_PCT-ni (rəqəm), ya da 'yoxdur'-u (mətn) qaytara bilər.
-- Bir sütunda həm rəqəm, həm mətn ola bilməz. Bu, CASE-dəki qayda ilə eynidir (6-cı bölmə).
-- Nəticədə mətn lazımdırsa, NVL2 işlədin (aşağıda).


-- ------------------------------------------------------------
-- 11.4 NVL2: boş olub-olmamasına görə iki fərqli dəyər
-- ------------------------------------------------------------

SELECT FIRST_NAME, COMMISSION_PCT,
       NVL2(COMMISSION_PCT, 'var', 'yoxdur') AS KOMISSIYA
FROM HR.EMPLOYEES;
-- Burada COMMISSION_PCT yalnız YOXLANILIR, qaytarılmır.
-- Qaytarılan iki dəyər ('var' və 'yoxdur') ikisi də mətndir, ona görə xəta yoxdur.

-- NVL2 rəqəmlə: komissiyası olanlara komissiya məbləği, olmayanlara 100 sabit bonus
SELECT FIRST_NAME, SALARY, COMMISSION_PCT,
       NVL2(COMMISSION_PCT, SALARY * COMMISSION_PCT, 100) AS BONUS
FROM HR.EMPLOYEES;
-- 2-ci və 3-cü dəyər ikisi də rəqəmdir.

-- SƏHV: 2-ci dəyər rəqəm, 3-cü dəyər mətndir. ORA-01722
-- SELECT NVL2(COMMISSION_PCT, COMMISSION_PCT, 'yoxdur') FROM HR.EMPLOYEES;


-- ------------------------------------------------------------
-- 11.5 Hansını nə vaxt işlətmək lazımdır
--
-- Boş yerə bir dəyər qoymaq lazımdırsa                 -> NVL
-- Boş olub-olmamasına görə iki fərqli dəyər lazımdırsa  -> NVL2
-- Şərt NULL yoxlamasından mürəkkəbdirsə (>, LIKE, IN)   -> CASE
-- ------------------------------------------------------------


-- ============================================================
-- 12. YADDA SAXLAYIN
--
-- - CASE ilə başlayır, END ilə bitir.
-- - Birinci doğru WHEN qalib gəlir: ən dar şərti yuxarıda yazın.
-- - ELSE yoxdursa, heç bir WHEN tutmayanda NULL çıxır.
-- - NULL-u IS NULL ilə yoxlayın, = NULL ilə yox. Simple CASE NULL-u tutmur.
-- - THEN və ELSE dəyərlərinin hamısı eyni tipdə olmalıdır.
-- - Alias tək dırnaqda yazılmır: AS MAAS_SEVIYYESI və ya AS "maas seviyyesi".
-- - CASE alias-ı ORDER BY-da işləyir, WHERE-də işləmir.
-- - NULL ilə hesab NULL verir. Boş dəyəri NVL ilə 0-a çevirin.
-- - NVL(x, y): x boşdursa y. NVL2(x, a, b): x doludursa a, boşdursa b.
-- - NVL və NVL2-də qaytarıla bilən dəyərlər eyni tipdə olmalıdır.
-- ============================================================
