-- ============================================================
-- SQL: FORMATLAR VƏ TARİXLƏR - QAYDALAR VƏ NÜMUNƏLƏR
-- Mövzular: || və CONCAT, mətn funksiyaları, rəqəm formatı (TO_CHAR),
--           TRUNC, tarix funksiyaları, TO_CHAR və TO_DATE ilə tarix formatları
-- Bütün nümunələr Oracle-da (FreeSQL) yazılıb.
-- HR cədvəllərinin adını həmişə HR. ilə yazın: HR.EMPLOYEES.
--
-- Tarix nəticələri bu faylda DD.MM.YYYY şəklində yazılıb.
-- FreeSQL tarixi ekranda öz formatında göstərə bilər (məsələn 15-APR-25), bu normaldır.
-- "ƏLAVƏ" qeydi olan hissələri bilmək faydalıdır, amma əsas mövzu deyil.
-- ============================================================


-- ============================================================
-- 1. || İLƏ BİRLƏŞDİRMƏ
--
-- || iki mətni yan-yana yapışdırır. Aralarında boşluq lazımdırsa,
-- onu özümüz ' ' kimi yazmalıyıq.
-- ============================================================

-- 1.1 Ad və soyadı birləşdirmək
SELECT FIRST_NAME || ' ' || LAST_NAME AS TAM_AD
FROM HR.EMPLOYEES;
-- Steven King

-- Boşluq yazılmasa:
SELECT FIRST_NAME || LAST_NAME AS TAM_AD
FROM HR.EMPLOYEES;
-- StevenKing


-- 1.2 Rəqəm də birləşir: Oracle onu özü mətnə çevirir
SELECT FIRST_NAME || ' ' || SALARY || ' AZN alir' AS MELUMAT
FROM HR.EMPLOYEES;
-- Steven 24000 AZN alir


-- 1.3 NULL ilə birləşdirmə
-- Oracle-da NULL birləşdirmədə boş mətn kimi davranır.
SELECT FIRST_NAME || ' - ' || COMMISSION_PCT AS MELUMAT
FROM HR.EMPLOYEES;
-- Steven -        (komissiya NULL-dur, heç nə yazılmır)
-- Başqa bazalarda (PostgreSQL, SQL Server) belə birləşdirmənin nəticəsi tam NULL olur.


-- 1.4 NULL əvəzinə mətn yazmaq: NVL + TO_CHAR
-- SƏHV: NVL-in iki dəyəri eyni tipdə olmalıdır (biri rəqəm, biri mətn). ORA-01722
SELECT FIRST_NAME || ' - ' || NVL(COMMISSION_PCT, 'yoxdur')
FROM HR.EMPLOYEES;

-- DÜZGÜN: əvvəl rəqəmi TO_CHAR ilə mətnə çeviririk
SELECT FIRST_NAME || ' - ' || NVL(TO_CHAR(COMMISSION_PCT), 'yoxdur') AS MELUMAT
FROM HR.EMPLOYEES;
-- Steven - yoxdur
-- Eleni - .2       <- diqqət: 0 itib (bölmə 1.5)


-- 1.5 0.2 niyə .2 kimi çıxır
-- Rəqəm formatsız mətnə çevriləndə Oracle 0 ilə 1 arasındakı rəqəmlərin
-- önündəki 0-ı yazmır. Cədvəldəki dəyər dəyişmir, yalnız mətn belə görünür.
SELECT TO_CHAR(0.2)  FROM DUAL;   -- '.2'
SELECT TO_CHAR(0.15) FROM DUAL;   -- '.15'
SELECT TO_CHAR(1.5)  FROM DUAL;   -- '1.5'  (1-dən böyüklərdə problem yoxdur)

-- Həll 1: TO_CHAR-a format vermək (bölmə 5)
SELECT FIRST_NAME || ' - ' || NVL(TO_CHAR(COMMISSION_PCT, 'FM0.00'), 'yoxdur') AS MELUMAT
FROM HR.EMPLOYEES;
-- Steven - yoxdur
-- Eleni - 0.20

-- Həll 2: faiz kimi göstərmək (0.2 * 100 = 20, sıfır problemi yoxdur)
SELECT FIRST_NAME || ' - ' || COMMISSION_PCT * 100 || '%' AS KOMISSIYA
FROM HR.EMPLOYEES
WHERE COMMISSION_PCT IS NOT NULL;
-- Eleni - 20%


-- ============================================================
-- 2. CONCAT
--
-- CONCAT(a, b) bir dəfədə YALNIZ 2 dəyər birləşdirir.
-- Çox dəyər üçün CONCAT-ı bir-birinin içinə yazırıq.
-- Qayda: n dəyər üçün n-1 CONCAT lazımdır.
-- İçdən çölə doğru oxunur.
-- ============================================================

-- 2.1 İki dəyər
SELECT CONCAT(FIRST_NAME, LAST_NAME) FROM HR.EMPLOYEES;
-- StevenKing

-- 2.2 Üç dəyər: ad + boşluq + soyad (2 CONCAT)
SELECT CONCAT(CONCAT(FIRST_NAME, ' '), LAST_NAME) AS TAM_AD
FROM HR.EMPLOYEES;
-- 1) CONCAT(FIRST_NAME, ' ')   -> 'Steven '
-- 2) CONCAT('Steven ', LAST_NAME) -> 'Steven King'

-- 2.3 Altı dəyər: 'Steven King (AD_PRES)' (5 CONCAT)
SELECT CONCAT(CONCAT(CONCAT(CONCAT(CONCAT(FIRST_NAME, ' '), LAST_NAME), ' ('), JOB_ID), ')') AS MELUMAT
FROM HR.EMPLOYEES;
-- Addım-addım:
-- CONCAT(FIRST_NAME, ' ')   -> 'Steven '
-- CONCAT(..., LAST_NAME)    -> 'Steven King'
-- CONCAT(..., ' (')         -> 'Steven King ('
-- CONCAT(..., JOB_ID)       -> 'Steven King (AD_PRES'
-- CONCAT(..., ')')          -> 'Steven King (AD_PRES)'

-- Eyni şey || ilə:
SELECT FIRST_NAME || ' ' || LAST_NAME || ' (' || JOB_ID || ')' AS MELUMAT
FROM HR.EMPLOYEES;

-- 2.4 CONCAT ilə || birlikdə
SELECT CONCAT(FIRST_NAME || ' ', LAST_NAME) AS TAM_AD
FROM HR.EMPLOYEES;
-- FIRST_NAME || ' ' bir dəyər sayılır, CONCAT-a cəmi 2 dəyər gedir.

-- 2.5 Rəqəm, NULL və tarix ilə
SELECT CONCAT(CONCAT(FIRST_NAME, ': '), TO_CHAR(SALARY, 'FM99,999')) AS MAAS,
       CONCAT(CONCAT(FIRST_NAME, ' - '), NVL(TO_CHAR(COMMISSION_PCT, 'FM0.00'), 'yoxdur')) AS KOMISSIYA,
       CONCAT(CONCAT(FIRST_NAME, ' ise girib: '), TO_CHAR(HIRE_DATE, 'DD.MM.YYYY')) AS TARIX
FROM HR.EMPLOYEES;
-- Steven: 24,000 | Steven - yoxdur | Steven ise girib: 17.06.....

-- 2.6 Səhvlər
-- 3 dəyəri birbaşa yazmaq: ORA-00909 invalid number of arguments
SELECT CONCAT(FIRST_NAME, ' ', LAST_NAME) FROM HR.EMPLOYEES;

-- İç-içə CONCAT-da mötərizəni bağlamağı unutmaq: ORA-00907 missing right parenthesis
SELECT CONCAT(CONCAT(FIRST_NAME, ' '), LAST_NAME FROM HR.EMPLOYEES;

-- Nəticə: CONCAT-ı bilmək lazımdır, amma 3 və daha çox dəyər olanda || daha rahatdır.

-- 4. MƏTN FUNKSİYALARI

-- 4.1 Böyük-kiçik hərf
SELECT UPPER(FIRST_NAME)        AS BOYUK,      -- STEVEN
       LOWER(FIRST_NAME)        AS KICIK,      -- steven
       INITCAP('sTEVEN kING')   AS ILK_HERF    -- Steven King (hər sözün ilk hərfi böyük)

-- 4.2 LENGTH: mətnin uzunluğu (boşluqlar da sayılır)
SELECT FIRST_NAME, LENGTH(FIRST_NAME) AS UZUNLUQ FROM HR.EMPLOYEES;
-- Steven 6

-- 4.3 SUBSTR(mətn, başlanğıc, uzunluq): mətnin bir hissəsini götürmək
SELECT SUBSTR('Steven', 1, 3) FROM DUAL;   -- Ste  (1-ci simvoldan başlayıb 3 simvol)
SELECT SUBSTR('Steven', 3)    FROM DUAL;   -- even (uzunluq yazılmasa, sona qədər)
SELECT SUBSTR('Steven', -2)   FROM DUAL;   -- en   (mənfi başlanğıc: sondan sayılır)
-- Oracle-da sayma 1-dən başlayır, 0-dan yox.

-- 4.4 INSTR(mətn, axtarılan): simvolun neçənci yerdə olduğunu tapır
SELECT INSTR('Steven', 'v') FROM DUAL;     -- 4
SELECT INSTR('Steven', 'x') FROM DUAL;     -- 0 (tapılmadı)

-- 4.5 REPLACE(mətn, nə, nə ilə): əvəz etmək
SELECT REPLACE('515.123.4567', '.', '-') FROM DUAL;   -- 515-123-4567
SELECT REPLACE('515.123.4567', '.', '')  FROM DUAL;   -- 5151234567 (silmək)

-- 4.6 TRIM, LTRIM, RTRIM: boşluqları silmək
SELECT '[' || TRIM('   salam   ')  || ']' FROM DUAL;  -- [salam]      (hər iki tərəfdən)
SELECT '[' || LTRIM('   salam   ') || ']' FROM DUAL;  -- [salam   ]   (yalnız soldan)
SELECT '[' || RTRIM('   salam   ') || ']' FROM DUAL;  -- [   salam]   (yalnız sağdan)
-- [ ] mötərizələri boşluqları gözlə görmək üçün yazılıb.

-- 4.7 LPAD, RPAD: mətni müəyyən uzunluğa qədər doldurmaq
SELECT LPAD(SALARY, 8, '*') FROM HR.EMPLOYEES;     -- ***24000 (soldan * ilə 8 simvola qədər)
SELECT RPAD(FIRST_NAME, 10, '.') FROM HR.EMPLOYEES; -- Steven....
SELECT LPAD(EMPLOYEE_ID, 5, '0') FROM HR.EMPLOYEES; -- 00100

FROM HR.EMPLOYEES;
-- sking@company.com

-- 5. RƏQƏM FORMATI: TO_CHAR(rəqəm, 'format')
--
-- Format simvolları:
-- 9    bu yerdə rəqəm VARSA yazılsın
-- 0    bu yerdə rəqəm MÜTLƏQ yazılsın (yoxdursa 0 yazılır)
-- ,    minlik ayırıcı
-- .    onluq nöqtə
-- $    dollar işarəsi
-- FM   öndəki boşluğu silir (və 9-la yazılan kəsrin sonundakı artıq 0-ları)
--
-- TO_CHAR-ın nəticəsi MƏTNDİR, rəqəm deyil.
-- ============================================================

SELECT SALARY,
       TO_CHAR(SALARY, '99,999')      AS F1,   -- ' 24,000' (önündə boşluq: mənfi işarəsi üçün yer)
       TO_CHAR(SALARY, 'FM99,999')    AS F2,   -- '24,000'
       TO_CHAR(SALARY, 'FM99,999.00') AS F3,   -- '24,000.00'
       TO_CHAR(SALARY, 'FM$99,999')   AS F4    -- '$24,000'
FROM HR.EMPLOYEES;

-- Boşluğu gözlə görmək üçün:
SELECT '[' || TO_CHAR(24000, '99,999')   || ']' FROM DUAL;   -- [ 24,000]
SELECT '[' || TO_CHAR(24000, 'FM99,999') || ']' FROM DUAL;   -- [24,000]

-- 0 ilə 9-un fərqi
SELECT TO_CHAR(0.2,  'FM0.00') FROM DUAL;   -- 0.20
SELECT TO_CHAR(0.2,  'FM0.99') FROM DUAL;   -- 0.2
SELECT TO_CHAR(0.15, 'FM0.99') FROM DUAL;   -- 0.15
SELECT TO_CHAR(7,    'FM000')  FROM DUAL;   -- 007

-- Format maskası rəqəmdən kiçikdirsə, nəticə #### olur
SELECT TO_CHAR(24000, '999') FROM DUAL;     -- ####

-- TƏLƏ: TO_CHAR nəticəsi mətndir, mətn kimi sıralanır
SELECT FIRST_NAME, TO_CHAR(SALARY, 'FM99,999') AS MAAS
FROM HR.EMPLOYEES
ORDER BY MAAS DESC;
-- '9,000' '24,000'-dən yuxarıda çıxır, çünki mətn kimi '9' > '2'

-- DÜZGÜN: göstərmək üçün formatlayırıq, sıralamaq üçün rəqəmin özünü işlədirik
SELECT FIRST_NAME, TO_CHAR(SALARY, 'FM99,999') AS MAAS
FROM HR.EMPLOYEES
ORDER BY SALARY DESC;

-- MOD(a, b): a-nı b-yə böləndə qalan qalıq
SELECT MOD(10, 3), MOD(150, 12), MOD(24, 12) FROM DUAL;   -- 1, 6, 0


-- ============================================================
-- 6. TRUNC: KƏSMƏK
--
-- TRUNC verilən yerdən sonrakı hissəni ATIR, yuvarlaqlaşdırmır.
-- Rəqəmdə də, tarixdə də işləyir (tarix üçün bölmə 12).
-- ============================================================

SELECT TRUNC(4.789)    FROM DUAL;   -- 4      (kəsr hissəsi tam silinir)
SELECT TRUNC(4.789, 1) FROM DUAL;   -- 4.7    (vergüldən sonra 1 rəqəm qalır)
SELECT TRUNC(4.789, 2) FROM DUAL;   -- 4.78   (vergüldən sonra 2 rəqəm qalır)
SELECT TRUNC(1789, -2) FROM DUAL;   -- 1700   (mənfi: sağdan 2 rəqəm sıfırlanır)

-- TRUNC və ROUND
--                TRUNC    ROUND
-- 4.789          4        5
-- 4.789, 2       4.78     4.79
-- 1789, -2       1700     1800
-- TRUNC sonrakı rəqəmə baxmır, sadəcə kəsir. ROUND baxır və yuvarlaqlaşdırır.

-- TRUNC və FLOOR: fərq yalnız mənfi rəqəmlərdə görünür
SELECT TRUNC(-4.7), FLOOR(-4.7) FROM DUAL;   -- -4, -5
-- TRUNC kəsri atır (sıfıra doğru gedir), FLOOR həmişə aşağı (kiçik rəqəmə) gedir.


-- ============================================================
-- 7. TARİX: DATE TİPİ VƏ BUGÜNKÜ TARİX
--
-- Oracle-da DATE tipinin içində tarixdən əlavə SAAT, DƏQİQƏ və SANİYƏ də var,
-- ekranda görünməsə belə.
-- ============================================================

SELECT SYSDATE        FROM DUAL;   -- bugünkü tarix + saat (serverin saatı)
SELECT CURRENT_DATE   FROM DUAL;   -- sessiyanın saat qurşağına görə tarix + saat
SELECT SYSTIMESTAMP   FROM DUAL;   -- ƏLAVƏ: saniyənin hissələri və saat qurşağı ilə
SELECT TRUNC(SYSDATE) FROM DUAL;   -- bugünkü tarix, saat 00:00:00

-- Saatı görmək üçün:
SELECT TO_CHAR(SYSDATE, 'DD.MM.YYYY HH24:MI:SS')        FROM DUAL;   -- 14.11.2026 15:42:18
SELECT TO_CHAR(TRUNC(SYSDATE), 'DD.MM.YYYY HH24:MI:SS') FROM DUAL;   -- 14.11.2026 00:00:00

-- Qeyd: SYSDATE serverin saatıdır. FreeSQL-in serveri başqa ölkədədir,
-- ona görə saat Bakı saatından fərqli ola bilər.


-- ============================================================
-- 8. TARİXLƏ HESABLAMA
--
-- tarix + rəqəm  = tarix (rəqəm qədər gün sonra)
-- tarix - rəqəm  = tarix (rəqəm qədər gün əvvəl)
-- tarix - tarix  = rəqəm (aradakı gün sayı)
-- tarix + tarix  = XƏTA
-- ============================================================

SELECT DATE '2025-01-15' + 10 FROM DUAL;                  -- 25.01.2025
SELECT DATE '2025-01-15' - 20 FROM DUAL;                  -- 26.12.2024
SELECT DATE '2025-03-01' - DATE '2025-02-01' FROM DUAL;   -- 28
SELECT SYSDATE + 1/24 FROM DUAL;                          -- ƏLAVƏ: 1 saat sonra (1 gün / 24)

-- SƏHV: iki tarixi toplamaq olmaz. ORA-00975
SELECT DATE '2025-01-15' + DATE '2025-01-01' FROM DUAL;

-- İşçi neçə gündür işləyir
SELECT FIRST_NAME,
       SYSDATE - HIRE_DATE        AS GUN,        -- kəsrli: 7820.65593
       TRUNC(SYSDATE - HIRE_DATE) AS TAM_GUN     -- 7820
FROM HR.EMPLOYEES;
-- Kəsr ona görə çıxır ki, SYSDATE-də saat var, HIRE_DATE-də isə saat 00:00-dır.


-- ============================================================
-- 9. MONTHS_BETWEEN(tarix1, tarix2): İKİ TARİX ARASINDA NEÇƏ AY VAR
--
-- tarix1 - tarix2 hesablanır, nəticə aylarla.
-- ============================================================

SELECT MONTHS_BETWEEN(DATE '2025-03-15', DATE '2025-01-15') FROM DUAL;  --  2
SELECT MONTHS_BETWEEN(DATE '2025-01-15', DATE '2025-03-15') FROM DUAL;  -- -2
SELECT MONTHS_BETWEEN(DATE '2025-03-20', DATE '2025-03-05') FROM DUAL;  --  0.4838...
SELECT MONTHS_BETWEEN(DATE '2025-03-31', DATE '2025-02-28') FROM DUAL;  --  1

-- Nüanslar:
-- - Sıra vacibdir. Böyük (gec) tarix birinci yazılmasa, nəticə mənfi çıxır.
-- - Nəticə adətən kəsrli olur. Kəsr ayı 31 gün sayaraq hesablanır: 15 gün / 31 = 0.48
-- - Nəticə tam ədəd iki halda çıxır:
--   hər iki tarix ayın eyni günüdür (15 və 15),
--   və ya hər iki tarix ayın son günüdür (31 mart və 28 fevral).
-- - Tam ay sayı üçün TRUNC(MONTHS_BETWEEN(...)) yazılır.

-- 9.1 Yaş: TRUNC(ay sayı / 12)
SELECT TRUNC(MONTHS_BETWEEN(DATE '2025-11-14', DATE '2000-11-15') / 12) FROM DUAL;
-- 24 (düzgün: ad günü sabahdır, hələ 25 tamam olmayıb)

-- SƏHV yol: gün sayını 365-ə bölmək
SELECT TRUNC((DATE '2025-11-14' - DATE '2000-11-15') / 365) FROM DUAL;
-- 25 (səhv: uzun illərin artıq günləri nəticəni dəyişir)

-- 9.2 Staj: neçə il və neçə ay
SELECT FIRST_NAME || ' ' || LAST_NAME                     AS ISCI,
       TO_CHAR(HIRE_DATE, 'DD.MM.YYYY')                   AS ISE_GIRIS,
       TRUNC(MONTHS_BETWEEN(SYSDATE, HIRE_DATE))          AS CEMI_AY,
       TRUNC(MONTHS_BETWEEN(SYSDATE, HIRE_DATE) / 12)     AS STAJ_IL,
       MOD(TRUNC(MONTHS_BETWEEN(SYSDATE, HIRE_DATE)), 12) AS STAJ_AY
FROM HR.EMPLOYEES;
-- Məsələn CEMI_AY = 150 olarsa: STAJ_IL = 12, STAJ_AY = MOD(150, 12) = 6, yəni 12 il 6 ay.
-- Burada ROUND yox, TRUNC lazımdır: 22.9 il işləyən adamın stajı 22 ildir, 23 yox.


-- ============================================================
-- 10. ADD_MONTHS(tarix, n): n AY ƏLAVƏ ETMƏK
-- ============================================================

SELECT ADD_MONTHS(DATE '2025-01-15', 3)  FROM DUAL;   -- 15.04.2025
SELECT ADD_MONTHS(DATE '2025-01-15', -1) FROM DUAL;   -- 15.12.2024  (mənfi = geriyə)
SELECT ADD_MONTHS(DATE '2025-01-15', 12) FROM DUAL;   -- 15.01.2026  (1 il = 12 ay)

-- Ayın sonu ilə bağlı nüanslar
SELECT ADD_MONTHS(DATE '2025-01-31', 1)  FROM DUAL;   -- 28.02.2025  (31 fevral yoxdur -> ayın son günü)
SELECT ADD_MONTHS(DATE '2025-02-28', 1)  FROM DUAL;   -- 31.03.2025  (!) 28 mart yox
SELECT ADD_MONTHS(DATE '2024-02-29', 12) FROM DUAL;   -- 28.02.2025
-- Qayda: başlanğıc tarix ayın SON GÜNÜDÜRSƏ, nəticə də ayın SON GÜNÜ olur.
-- 28 fevral 2025 fevralın son günüdür, ona görə +1 ay = martın son günü = 31 mart.

-- 3 aylıq sınaq müddəti nə vaxt bitib
SELECT FIRST_NAME, HIRE_DATE, ADD_MONTHS(HIRE_DATE, 3) AS SINAQ_BITIR
FROM HR.EMPLOYEES;

-- ============================================================
-- 12. TARİXLƏRDƏ TRUNC VƏ ROUND
--
-- TRUNC(tarix)          saatı silir -> 00:00:00
-- TRUNC(tarix, 'MM')    gün və saatı silir -> ayın ilk günü
-- TRUNC(tarix, 'Q')     rübün ilk günü
-- TRUNC(tarix, 'YYYY')  ilin ilk günü
-- Format verilsə, həmin vahiddən KİÇİK olan hər şey sıfırlanır.
-- ============================================================

SELECT TRUNC(DATE '2025-08-20', 'MM')   FROM DUAL;   -- 01.08.2025
SELECT TRUNC(DATE '2025-08-20', 'Q')    FROM DUAL;   -- 01.07.2025
SELECT TRUNC(DATE '2025-08-20', 'YYYY') FROM DUAL;   -- 01.01.2025

-- ƏLAVƏ: ROUND ən yaxın dövrə yuvarlaqlaşdırır
SELECT ROUND(DATE '2025-03-10', 'MM')   FROM DUAL;   -- 01.03.2025
SELECT ROUND(DATE '2025-03-16', 'MM')   FROM DUAL;   -- 01.04.2025  (16-sı və sonrası yuxarı)
SELECT ROUND(DATE '2025-07-01', 'YYYY') FROM DUAL;   -- 01.01.2026  (1 iyuldan sonra yuxarı)

-- TRUNC nə vaxt lazımdır:
-- WHERE HIRE_DATE = SYSDATE          heç vaxt tapmır: SYSDATE-də saat 15:42:18 var
-- WHERE HIRE_DATE = TRUNC(SYSDATE)   düzgün: ikisi də 00:00:00


-- ============================================================
-- 13. EXTRACT: TARİXDƏN BİR HİSSƏNİ RƏQƏM KİMİ GÖTÜRMƏK
-- ============================================================

-- 14. TO_CHAR(tarix, 'format'): TARİXİ İSTƏDİYİN FORMADA GÖSTƏRMƏK

-- Format       Nə göstərir                          20.08.2025 14:30:05 üçün
-- YYYY         4 rəqəmli il                         2025
-- YY           2 rəqəmli il                         25
-- YEAR         il sözlə (ƏLAVƏ)                     TWENTY TWENTY-FIVE
-- Q            rüb                                  3
-- MM           ay nömrəsi                           08
-- MON / Mon    ayın qısa adı                        AUG / Aug
-- MONTH/Month  ayın tam adı (9 simvola qədər)       AUGUST / August
-- DD           ayın günü                            20
-- DDD          ilin neçənci günü                    232
-- DY / Dy      həftənin günü, qısa                  WED / Wed
-- DAY / Day    həftənin günü, tam (9 simvola qədər) WEDNESDAY / Wednesday
-- D            həftənin günü nömrə ilə (ƏLAVƏ, ölkə ayarından asılıdır)
-- WW / IW      ilin neçənci həftəsi (ƏLAVƏ)         34
-- HH24 / HH    saat (24 / 12 saatlıq)               14 / 02
-- MI           dəqiqə (MM DEYİL!)                   30
-- SS           saniyə                               05
-- AM           AM / PM                              PM
-- FM           artıq boşluqları və öndəki 0-ları silir
-- "mətn"       dırnaq içində istənilən söz
-- th           sıra şəkilçisi, ingiliscə (ƏLAVƏ)    20th
-- ============================================================

SELECT TO_CHAR(DATE '2025-08-05', 'DD.MM.YYYY')      FROM DUAL;   -- 05.08.2025

-- 17. TARİXƏ GÖRƏ QRUPLAŞDIRMA

-- 17.1 İllər üzrə işə qəbul
SELECT EXTRACT(YEAR FROM HIRE_DATE) AS IL, COUNT(*) AS SAY
FROM HR.EMPLOYEES
GROUP BY EXTRACT(YEAR FROM HIRE_DATE)
ORDER BY IL;

-- 17.2 İl-ay üzrə: 'YYYY-MM' mətn olsa da düzgün sıralanır
SELECT TO_CHAR(HIRE_DATE, 'YYYY-MM') AS IL_AY, COUNT(*) AS SAY
FROM HR.EMPLOYEES
GROUP BY TO_CHAR(HIRE_DATE, 'YYYY-MM')
ORDER BY IL_AY;

-- 17.3 TƏLƏ: ay adına görə sıralama əlifba sırası ilə gedir
SELECT TO_CHAR(HIRE_DATE, 'FMMonth') AS AY, COUNT(*) AS SAY
FROM HR.EMPLOYEES
GROUP BY TO_CHAR(HIRE_DATE, 'FMMonth')
ORDER BY AY;
-- April, August, December, ... (təqvim sırası ilə yox)

-- DÜZGÜN: ay nömrəsini də qruplaşdırmaya əlavə edib ona görə sıralamaq
SELECT TO_CHAR(HIRE_DATE, 'MM') AS AY_NO, TO_CHAR(HIRE_DATE, 'FMMonth') AS AY, COUNT(*) AS SAY
FROM HR.EMPLOYEES
GROUP BY TO_CHAR(HIRE_DATE, 'MM'), TO_CHAR(HIRE_DATE, 'FMMonth')
ORDER BY AY_NO;

-- 17.4 Həftənin hansı günü daha çox işə qəbul olub
SELECT TO_CHAR(HIRE_DATE, 'FMDay') AS GUN, COUNT(*) AS SAY
FROM HR.EMPLOYEES
GROUP BY TO_CHAR(HIRE_DATE, 'FMDay')
ORDER BY SAY DESC;

-- 20. YADDA SAXLAYIN
--
-- - || mətni birləşdirir. Boşluğu özünüz yazın: FIRST_NAME || ' ' || LAST_NAME
-- - Oracle-da NULL birləşdirmədə boş mətn kimi davranır.
-- - CONCAT yalnız 2 dəyər alır. n dəyər üçün n-1 CONCAT lazımdır.
-- - NVL-də rəqəm ilə mətn qarışmır: NVL(TO_CHAR(rəqəm), 'mətn').
-- - Formatsız TO_CHAR 0.2-ni '.2' edir. Format verin: 'FM0.00'.
-- - TO_CHAR-ın nəticəsi mətndir: göstərmək üçün formatlayın, sıralamaq üçün əsl dəyəri işlədin.
-- - TRUNC kəsir, ROUND yuvarlaqlaşdırır. Staj və yaş üçün TRUNC.
-- - DATE-in içində saat da var. Bu günlə müqayisə üçün TRUNC(SYSDATE).
-- - tarix - tarix = gün sayı. Ay üçün MONTHS_BETWEEN (gec tarix birinci).
-- - Ay əlavə etmək üçün ADD_MONTHS. Ayın son günü qaydasını unutmayın.
-- - EXTRACT rəqəm qaytarır, TO_CHAR mətn qaytarır.
-- - MM ay, MI dəqiqədir.
-- - Month və Day formatlarını FM ilə yazın, yoxsa sonda boşluqlar qalır.
-- - Mətni tarixə TO_DATE və ya DATE 'YYYY-MM-DD' ilə çevirin, tarixi mətnlə müqayisə etməyin.
-- - İli həmişə 4 rəqəmlə (YYYY) yazın.
-- ============================================================
