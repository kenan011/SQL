-- ============================================================
-- SQL DƏRS 4: QAYDALAR VƏ NÜMUNƏLƏR
-- Mövzular: SELF JOIN, IN ilə subquery
-- Bütün nümunələr Oracle-da (FreeSQL) yazılıb.
-- HR cədvəllərinin adını həmişə HR. ilə yazın: HR.EMPLOYEES.
-- ============================================================


-- ============================================================
-- 1. SELF JOIN NƏDİR
--
-- SELF JOIN cədvəlin ÖZ-ÖZÜ İLƏ birləşdirilməsidir.
-- Ayrıca SELF JOIN əmri YOXDUR. Adi JOIN və ya LEFT JOIN yazılır,
-- sadəcə eyni cədvəl iki dəfə, iki FƏRQLİ alias ilə yazılır.
--
-- Nə vaxt lazımdır: cədvəlin bir sətri eyni cədvəlin başqa sətrinə istinad edəndə.
-- Məsələn, işçinin meneceri də elə həmin cədvəldə olan bir işçidir.
-- ============================================================

-- Nümunə cədvəl. Əgər hələ yaratmamısınızsa, bu hissəni bütövlükdə işlədin.
CREATE TABLE demo_staff (
    employee_id NUMBER PRIMARY KEY,
    first_name  VARCHAR2(30),
    manager_id  NUMBER,
    salary      NUMBER
);

INSERT INTO demo_staff VALUES (1, 'Kamran', NULL, 5000);
INSERT INTO demo_staff VALUES (2, 'Aysel',  1,    3000);
INSERT INTO demo_staff VALUES (3, 'Murad',  1,    3500);
INSERT INTO demo_staff VALUES (4, 'Nigar',  3,    4000);
INSERT INTO demo_staff VALUES (5, 'Leyla',  3,    2500);

COMMIT;

-- Xəta çıxsa:
-- ORA-00955  cədvəl artıq var. Yenidən CREATE etməyə ehtiyac yoxdur.
-- ORA-00001  INSERT ikinci dəfə işlədilib, sətirlər artıq cədvəldədir.

-- Struktur:
-- Kamran direktordur, onun meneceri yoxdur (MANAGER_ID boşdur).
-- Aysel və Murad Kamran-a tabedir (MANAGER_ID = 1).
-- Nigar və Leyla Murada tabedir (MANAGER_ID = 3).
--
-- MANAGER_ID sütununda menecerin ADI yox, NÖMRƏSİ (EMPLOYEE_ID-si) saxlanılır.
-- Nigar-ın MANAGER_ID-si 3-dür: onun meneceri EMPLOYEE_ID-si 3 olan işçidir, yəni Murad.


-- ============================================================
-- 2. İŞÇİ VƏ MENECERİ
--
-- Cədvəlin iki surətini təsəvvür edin:
-- e  işçilər surəti
-- m  menecerlər surəti
-- İşçinin MANAGER_ID-sini menecerin EMPLOYEE_ID-si ilə tutuşdururuq.
-- ============================================================

SELECT e.first_name AS isci, m.first_name AS menecer
FROM demo_staff e
JOIN demo_staff m ON e.manager_id = m.employee_id;

-- ISCI   MENECER
-- Aysel  Kamran
-- Murad  Kamran
-- Nigar  Murad
-- Leyla  Murad

-- Oracle bunu necə edir: e-nin hər sətri üçün m-də EMPLOYEE_ID-si
-- həmin MANAGER_ID-yə bərabər olan sətri axtarır.
--
-- e (işçi)  e.manager_id  m-də tapılan
-- Kamran    (null)        heç nə, NULL heç nəyə bərabər deyil  -> düşür
-- Aysel     1             employee_id = 1 -> Kamran
-- Murad     1             employee_id = 1 -> Kamran
-- Nigar     3             employee_id = 3 -> Murad
-- Leyla     3             employee_id = 3 -> Murad

-- Diqqət: Kamran İŞÇİ kimi (ISCI sütununda) görünmür, çünki onun meneceri yoxdur.
-- Amma MENECER kimi görünür, çünki Aysel və Murad ona tabedir.


-- ------------------------------------------------------------
-- 2.1 LEFT JOIN: meneceri olmayan işçi də görünsün. 5 sətir.
-- ------------------------------------------------------------

SELECT e.first_name AS isci, m.first_name AS menecer
FROM demo_staff e
LEFT JOIN demo_staff m ON e.manager_id = m.employee_id;

-- ISCI    MENECER
-- Kamran  (null)       <- meneceri tapılmadı, amma soldadır, ona görə qalır
-- Aysel   Kamran
-- Murad   Kamran
-- Nigar   Murad
-- Leyla   Murad


-- ------------------------------------------------------------
-- 2.2 ON-un istiqaməti mənanı müəyyən edir
-- ------------------------------------------------------------

-- e.manager_id = m.employee_id  belə oxunur: "işçinin meneceri = menecerin nömrəsi"
--
-- Tərəfləri dəyişsəniz, xəta çıxmır, amma məna tərsinə çevrilir:
SELECT e.first_name AS isci, m.first_name AS menecer
FROM demo_staff e
JOIN demo_staff m ON e.employee_id = m.manager_id;
-- Kamran Aysel, Kamran Murad, Murad Nigar, Murad Leyla
-- İndi "MENECER" sütununda əslində TABE işçilər görünür.


-- ============================================================
-- 3. SELF JOIN İLƏ SÜZMƏK VƏ MÜQAYİSƏ ETMƏK
-- ============================================================

-- 3.1 Murada tabe olanlar
SELECT e.first_name
FROM demo_staff e
JOIN demo_staff m ON e.manager_id = m.employee_id
WHERE m.first_name = 'Murad';
-- Nigar, Leyla

-- 3.2 Menecerindən çox maaş alanlar
SELECT e.first_name, e.salary, m.first_name AS menecer, m.salary AS menecer_maas
FROM demo_staff e
JOIN demo_staff m ON e.manager_id = m.employee_id
WHERE e.salary > m.salary;
-- Nigar 4000, Murad 3500

-- Bunu sadə WHERE ilə yazmaq olmur. WHERE bir anda YALNIZ BİR sətrə baxır.
-- Nigar-ın sətrində onun öz maaşı (4000) və menecerinin NÖMRƏSİ (3) var,
-- amma menecerin MAAŞI (3500) yoxdur, o, Muradın sətrindədir.
-- SELF JOIN menecerin sətrini işçinin sətrinin yanına gətirir:
--
-- E.FIRST_NAME  E.SALARY  M.FIRST_NAME  M.SALARY
-- Aysel         3000      Kamran        5000
-- Murad         3500      Kamran        5000
-- Nigar         4000      Murad         3500     <- yalnız burada e.salary > m.salary
-- Leyla         2500      Murad         3500


-- ============================================================
-- 4. SELF JOIN + GROUP BY: hər menecerin neçə tabesi var
-- ============================================================

SELECT m.first_name AS menecer, COUNT(*) AS tabe_sayi
FROM demo_staff e
JOIN demo_staff m ON e.manager_id = m.employee_id
GROUP BY m.employee_id, m.first_name;
-- Kamran 2, Murad 2

-- GROUP BY-a m.employee_id də yazılır, çünki iki menecerin adı eyni ola bilər.


-- ============================================================
-- 5. SELF JOIN-DƏ TEZ-TEZ EDİLƏN SƏHVLƏR
-- ============================================================

-- SƏHV 1: SELF JOIN deyə əmr yoxdur. Sintaksis xətası verir.
-- SELECT ... FROM demo_staff e SELF JOIN demo_staff m ON ...
-- DÜZGÜN: JOIN və ya LEFT JOIN yazılır.

-- SƏHV 2: alias yazılmayıb. EMPLOYEE_ID və MANAGER_ID hər iki surətdə var,
-- Oracle hansını nəzərdə tutduğunuzu bilmir. ORA-00918 xətası verir.
-- SELECT first_name FROM demo_staff
-- JOIN demo_staff ON manager_id = employee_id;
-- DÜZGÜN: iki fərqli alias (e, m) və hər sütunun qarşısında alias.

-- SƏHV 3: ON-da tərəflər dəyişib (2.2-yə baxın). Xəta yoxdur, nəticənin mənası səhvdir.


-- ============================================================
-- 6. SUBQUERY NƏDİR
--
-- Subquery sorğunun İÇİNDƏ yazılan başqa sorğudur. Mötərizədə yazılır.
-- Əvvəl İÇƏRİDƏKİ sorğu işləyir, onun nəticəsini xarici sorğu istifadə edir.
--
-- Bu dərsdə subquery-ni IN ilə istifadə edirik:
-- WHERE sutun IN (SELECT sutun FROM ... WHERE ...)
-- ============================================================

-- Sual: IT və HR şöbələrində kimlər işləyir? (demo cədvəllər)

-- Addım 1. Əvvəl içəridəki sorğunu AYRICA işlədin və nəticəsinə baxın:
SELECT department_id
FROM demo_departments
WHERE department_name IN ('IT', 'HR');
-- 10, 20

-- Addım 2. Bu rəqəmləri əl ilə yazmaq olardı:
SELECT first_name
FROM demo_employees
WHERE department_id IN (10, 20);

-- Addım 3. Rəqəmlərin yerinə sorğunun özünü yazırıq:
SELECT first_name
FROM demo_employees
WHERE department_id IN (SELECT department_id
                        FROM demo_departments
                        WHERE department_name IN ('IT', 'HR'));
-- Aysel, Murad, Nigar

-- Fərq: sabah şöbələrin nömrəsi dəyişsə, Addım 2-dəki sorğu səhv nəticə verəcək,
-- Addım 3-dəki isə yenə düzgün işləyəcək.


-- ------------------------------------------------------------
-- 6.1 HR üzərində nümunə
-- ------------------------------------------------------------

-- Sales və IT şöbələrində işləyən işçilər (şöbə nömrələrini bilmədən)
SELECT first_name, last_name, department_id
FROM hr.employees
WHERE department_id IN (SELECT department_id
                        FROM hr.departments
                        WHERE department_name IN ('Sales', 'IT'));

-- Vəzifəsinin adında 'Manager' sözü olan işçilər
SELECT first_name, last_name, job_id
FROM hr.employees
WHERE job_id IN (SELECT job_id
                 FROM hr.jobs
                 WHERE job_title LIKE '%Manager%');


-- ------------------------------------------------------------
-- 6.2 Eyni cədvəldə subquery: SELF JOIN-in əvəzi
-- ------------------------------------------------------------

-- Sual: kimlər menecerdir (ən azı bir tabesi var)?
SELECT first_name
FROM demo_staff
WHERE employee_id IN (SELECT manager_id FROM demo_staff);
-- Kamran, Murad

-- İçəridəki sorğu NULL, 1, 1, 3, 3 qaytarır.
-- Xarici sorğu EMPLOYEE_ID-si bu siyahıda olanları götürür: 1 Kamran, 3 Murad.
-- Siyahıda NULL olması IN-ə mane olmur.

-- Eyni sual SELF JOIN ilə:
SELECT m.first_name
FROM demo_staff e
JOIN demo_staff m ON e.manager_id = m.employee_id;
-- Kamran, Kamran, Murad, Murad  <- adlar TƏKRARLANIR, çünki hər tabe ayrıca cüt yaradır

-- Təkrarları silmək üçün DISTINCT lazımdır:
SELECT DISTINCT m.first_name
FROM demo_staff e
JOIN demo_staff m ON e.manager_id = m.employee_id;
-- Kamran, Murad

-- IN-də təkrar yaranmır: hər sətir ya siyahıdadır, ya da yox.


-- ============================================================
-- 7. SUBQUERY-DƏ TEZ-TEZ EDİLƏN SƏHVLƏR
-- ============================================================

-- SƏHV 1: IN-in içindəki sorğu yalnız BİR sütun qaytarmalıdır. ORA-00913
-- WHERE department_id IN (SELECT department_id, department_name FROM hr.departments)

-- SƏHV 2: = yalnız BİR dəyər gözləyir, sorğu isə bir neçə sətir qaytarır. ORA-01427
-- WHERE department_id = (SELECT department_id FROM hr.departments
--                        WHERE department_name IN ('Sales', 'IT'))
-- DÜZGÜN: = yerinə IN yazın.
-- Subquery tam olaraq bir sətir qaytarırsa, = də işləyir, amma IN həmişə təhlükəsizdir.

-- SƏHV 3: IN-in içində ORDER BY yazılmır. ORA-00907
-- WHERE department_id IN (SELECT department_id FROM hr.departments ORDER BY department_id)

-- SƏHV 4: mətn böyük-kiçik hərfə həssasdır. Xəta yoxdur, amma subquery boş qayıdır.
-- WHERE department_name = 'sales'   -- HR-də 'Sales' yazılıb, 'sales' tapılmır
-- Subquery-ni əvvəlcə ayrıca işlədib nəticəsinə baxmaq bu səhvi dərhal göstərir.


-- ============================================================
-- 8. SELF JOIN, YOXSA SUBQUERY?
--
-- Nəticədə HƏR İKİ tərəfin sütunları lazımdırsa  -> SELF JOIN
--   (məsələn, işçinin adı və menecerinin adı yan-yana)
-- Yalnız bir siyahıya görə SÜZMƏK lazımdırsa     -> IN subquery
--   (məsələn, "menecer olan işçilər")
-- ============================================================


-- ============================================================
-- 9. YADDA SAXLAYIN
--
-- - SELF JOIN ayrıca əmr deyil: eyni cədvəl iki dəfə, iki fərqli alias ilə.
-- - ON-un istiqaməti vacibdir: e.manager_id = m.employee_id.
-- - INNER JOIN meneceri olmayanı atır, LEFT JOIN saxlayır.
-- - SELF JOIN iki fərqli sətri müqayisə etmək üçündür.
-- - Subquery mötərizədə yazılır və əvvəl işləyir.
-- - IN-in içindəki sorğu bir sütun qaytarır, ORDER BY yazılmır.
-- - Bir neçə dəyər gələ bilərsə, = yox, IN yazın.
-- - Subquery-ni əvvəlcə ayrıca işlədib yoxlayın.
-- ============================================================
